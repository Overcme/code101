```js
// Запрашиваем имя у пользователя
let userName = prompt("Как вас зовут?");

// Проверяем, ввел ли пользователь имя
if (userName) {
    // Выводим приветствие на экран
    document.write(`<h2>Привет, ${userName}! Добро пожаловать на сайт.</h2>`);
} else {
    document.write("<h2>Привет, Гость!</h2>");
}
```
